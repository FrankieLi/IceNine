# Task A, Step 0: fixed-orientation parity. Applied to a SCRATCH copy of the repo (CMakeLists.txt, Src/,
# XDM++/, CostFunctionBenchmark/), run from the scratch root; Src/ in the repository is never touched.
#   uv run python cpp_parity_patch.py.txt   (copy to patch.py first or run via `python -` stdin)
# With env PARITY_IN (binary records) and PARITY_OUT (text) set, the first call of
# DiscreteRefinement::RunDiscreteSearch (level 0 of the adaptive reconstruction) evaluates every record
# with three arms and exits: INT (the production LocalAngularInterpProcess path: scatter vectors and omegas
# from the FZ centre, then InterpolatedVoxelCostFunction with the local-grid matrix; pixel radius 3, centre
# point), EXS (exact omegas of the product orientation; pixel radius 3, centre point; same Q-max) and
# EXF (the complete cost: all reflections, triangle rasterisation). Per peak line `P ...` is printed from
# inside XDMSampleVertexOverlapCounter::CalculateDiffractionOverlap.
import re

def patch(path, pairs):
    s = open(path).read()
    for a, b in pairs:
        assert s.count(a) == 1, (path, a, s.count(a))
        s = s.replace(a, b)
    open(path, 'w').write(s)

DRIVER = r'''
    if( getenv( "PARITY_IN" ) != NULL )
    {
      // ---- Task A step 0 scratch driver: evaluate fixed orientations, write per-peak records, exit ----
      std::cout << "PARITY nQMax " << nQMax << " n_fz_in_call " << ( pEnd - pFirst ) << " n_local " << oLocalGrid.size() << std::endl;
      FILE * fin = fopen( getenv( "PARITY_IN" ), "rb" );
      FILE * fout = fopen( getenv( "PARITY_OUT" ), "w" );
      int nRec = 0;
      if( fread( &nRec, 4, 1, fin ) != 1 ) exit( 1 );
      CSample & oSampleRef = oSetup.SampleGeometry();
      const DetectorListT & oDetListRef = oSetup.ExperimentalSetup().GetDetectorList();
      const XDMSimulation::CSimulationRange & oRangeRef = oSetup.ExperimentalSetup().GetRangeToIndexMap();
      vector<CRecpVector> oAllRecips = oSampleRef.GetStructureList()[ oVoxel.nPhase ].GetReflectionVectorList();
      std::sort( oAllRecips.begin(), oAllRecips.end(),
                 [](const CRecpVector& a, const CRecpVector& b) { return a.fMag < b.fMag; } );
      vector<CRecpVector> oScreenRecips;
      for( size_t r = 0; r < oAllRecips.size(); r ++ )
        if( ! ( oAllRecips[ r ].fMag > (Float) nQMax ) ) oScreenRecips.push_back( oAllRecips[ r ] );
      fprintf( fout, "H nrecip_all %d nrecip_screen %d nQMax %d nrec %d\n", (int) oAllRecips.size(), (int) oScreenRecips.size(), (int) nQMax, nRec );
      {
        SVector3 c = oVoxel.GetCenter();
        fprintf( fout, "V center %a %a %a\n", (double) c.m_fX, (double) c.m_fY, (double) c.m_fZ );
        for( int q = 0; q < 3; q ++ )
          fprintf( fout, "V vertex%d %a %a %a\n", q, (double) oVoxel.pVertex[ q ].m_fX, (double) oVoxel.pVertex[ q ].m_fY, (double) oVoxel.pVertex[ q ].m_fZ );
      }
      for( int rec = 0; rec < nRec; rec ++ )
      {
        int kind, id; float fc[ 9 ], fl[ 9 ];
        if( fread( &kind, 4, 1, fin ) != 1 || fread( &id, 4, 1, fin ) != 1 || fread( fc, 4, 9, fin ) != 9 || fread( fl, 4, 9, fin ) != 9 ) exit( 2 );
        SMatrix3x3 oCenter, oLocal;
        for( int i = 0; i < 3; i ++ ) for( int j = 0; j < 3; j ++ ) { oCenter.m[ i ][ j ] = fc[ 3 * i + j ]; oLocal.m[ i ][ j ] = fl[ 3 * i + j ]; }
        SMatrix3x3 oProd = oLocal * oCenter;
        fprintf( fout, "O %d kind %d prod", id, kind );
        for( int i = 0; i < 3; i ++ ) for( int j = 0; j < 3; j ++ ) fprintf( fout, " %a", (double) oProd.m[ i ][ j ] );
        fprintf( fout, "\n" );
        // ---- arm INT: production interpolated screen ----
        {
          fprintf( fout, "A INT\n" ); PDOut() = fout;
          CostFunctions::SOverlapInfo oInfo; oInfo.Initialize();
          SearchDetails::ScatteringVectorListT oScat;
          HEDM::PrecomputeScatteringVector( oScat, oCenter, oDetListRef, oRangeRef, oScreenRecips.begin(), oScreenRecips.end(), oSimulator );
          fprintf( fout, "S nobs %d obsrefl", (int) oScat.size() );
          for( size_t q = 0; q < PDObsRefl().size(); q ++ ) fprintf( fout, " %d", PDObsRefl()[ q ] );
          fprintf( fout, "\n" );
          oInfo = VoxelCostFunction( oVoxel, oSampleRef, oDetListRef, oRangeRef, oScat, oLocal, oSetup.Data() );
          PDOut() = NULL;
          fprintf( fout, "R pixov %d pixon %d pkov %d pkon %d quality %a conf %a\n", (int) oInfo.nPixelOverlap, (int) oInfo.nPixelOnDetector,
                   (int) oInfo.nPeakOverlap, (int) oInfo.nPeakOnDetector, (double) oInfo.fQuality,
                   (double) CostFunctions::Utilities::GetConfidence( oInfo ) );
        }
        // ---- arm EXS: exact omegas, same screen (radius 3, centre) ----
        {
          fprintf( fout, "A EXS\n" ); PDOut() = fout;
          vector<SPeakInfo> oPeaks;
          oSimulator.GetObservablePeaks( oPeaks, oProd, oScreenRecips, oDetListRef );
          vector<SVector3> oVert = VoxelCostFunction.VoxelToVerticesFn( oVoxel );
          CostFunctions::SOverlapInfo oInfo = VoxelCostFunction.OverlapFn.CalculateDiffractionOverlap( oVert, oPeaks, oSampleRef, oDetListRef, oRangeRef, oSetup.Data() );
          PDOut() = NULL;
          fprintf( fout, "R pixov %d pixon %d pkov %d pkon %d quality %a conf %a\n", (int) oInfo.nPixelOverlap, (int) oInfo.nPixelOnDetector,
                   (int) oInfo.nPeakOverlap, (int) oInfo.nPeakOnDetector, (double) oInfo.fQuality,
                   (double) CostFunctions::Utilities::GetConfidence( oInfo ) );
        }
        // ---- arm EXF: complete cost (all reflections, triangle) ----
        {
          fprintf( fout, "A EXF\n" ); PDOut() = fout;
          SamplePointT oTmp = oVoxel; oTmp.oOrientMatrix = oProd;
          CostFunctions::SOverlapInfo oInfo = WrappedCostFn( oTmp );
          PDOut() = NULL;
          fprintf( fout, "R pixov %d pixon %d pkov %d pkon %d quality %a conf %a\n", (int) oInfo.nPixelOverlap, (int) oInfo.nPixelOnDetector,
                   (int) oInfo.nPeakOverlap, (int) oInfo.nPeakOnDetector, (double) oInfo.fQuality,
                   (double) CostFunctions::Utilities::GetConfidence( oInfo ) );
        }
        fprintf( fout, "E\n" );
      }
      fclose( fout );
      std::cout << "PARITY done " << nRec << std::endl;
      exit( 0 );
    }
'''
